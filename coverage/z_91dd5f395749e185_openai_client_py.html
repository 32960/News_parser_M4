24def get_openai_client() -> OpenAI: 

25 if not settings.openai_api_key: 

26 raise ValueError("OpenAI API key not configured") 

27 return OpenAI( 

28 api_key=settings.openai_api_key.get_secret_value(),

32def text_for_generation(news: NewsItem) -> str: 

33 """ 

34 Spec rule: send non-empty summary to AI if the source gave one, 

35 otherwise send raw_text. 

36 """ 

37 summary = (news.summary or "").strip() 

38 if summary: 

39 return summary 

40 return (news.raw_text or "").strip()

43def generate_text(text: str) -> str: 

44 """ 

45 Call OpenAI and return non-empty post text. 

46 Raises on API errors or empty model output (so Celery can mark generation_failed). 

47 """ 

48 if not text or not text.strip(): 

49 raise ValueError("Empty text passed to AI generation") 

50 

51 try: 

52 with get_openai_client() as client: 

53 response = client.responses.create( 

54 model=settings.openai_model, 

55 instructions=GENERATE_TEXT_PROMPT, 

56 input=text, 

57 ) 

58 output = (response.output_text or "").strip() 

59 if not output: 

60 raise ValueError("AI returned empty text") 

61 return output 

62 except RateLimitError: 

63 # Rate limit = too many requests to OpenAI in a short time (HTTP 429) 

64 logger.error("OpenAI rate limit exceeded while generating a post") 

65 raise 

66 except (ValueError, OpenAIError) as exc: 

67 # Do not log secrets (API key). Only the error type/message from the SDK. 

68 logger.error("Failed to generate text: %s", exc) 

69 raise
